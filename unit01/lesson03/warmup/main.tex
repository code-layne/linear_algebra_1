\begin{notesbox}{Getting Ready: Skills We'll Use Today}
\small These Lesson~1.1 moves become today's \emph{matrix} and \emph{column space}. Answer quickly.

\vspace{4pt}
\textbf{1. Linear combination.} Scale each vector, then add.
\begin{enumerate}[label=(\alph*), leftmargin=2.0em, itemsep=10pt]
  \item $2\begin{bmatrix} 1 \\ 2 \end{bmatrix} + 3\begin{bmatrix} 0 \\ 1 \end{bmatrix}
        = \blank{3.0cm}$
  \item $1\begin{bmatrix} 3 \\ 1 \end{bmatrix} + 2\begin{bmatrix} 1 \\ 2 \end{bmatrix}
        = \blank{3.0cm}$
\end{enumerate}

\vspace{8pt}
\textbf{2. Is it a multiple?} Is $\begin{bmatrix} 2 \\ 4 \end{bmatrix}$ a multiple of
$\begin{bmatrix} 1 \\ 2 \end{bmatrix}$? Is $\begin{bmatrix} 3 \\ 1 \end{bmatrix}$? \writeline

\vspace{6pt}
\textbf{3. Line or plane?} Two \emph{non-parallel} vectors in the plane --- do all their
combinations $c\vv+d\ww$ fill a single \emph{line} or the \emph{whole plane}? \writeline
\end{notesbox}
