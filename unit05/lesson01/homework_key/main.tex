\begin{notesbox}{Practice}
Expand along a row (pick one with a zero when you can). Recall: $|\det|$ is the \textbf{volume} of the
box the columns span, and a triangular matrix's determinant is the \emph{product of its diagonal}.

\begin{enumerate}[leftmargin=1.9em, itemsep=11pt]
  \item \textbf{Compute determinants.}
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=5pt]
      \item $\det\begin{bmatrix} 1 & 2 & 0 \\ 0 & 3 & 1 \\ 1 & 0 & 2 \end{bmatrix}$ \hfill \ans{$1(6)-2(-1)+0=8$}
      \item $\det\begin{bmatrix} 2 & 0 & 0 \\ 1 & 3 & 0 \\ 4 & 5 & 1 \end{bmatrix}$ \; {\footnotesize(triangular)} \hfill \ans{$2\cdot3\cdot1=6$}
      \item $\det\begin{bmatrix} 1 & 2 & 3 \\ 2 & 4 & 6 \\ 0 & 1 & 1 \end{bmatrix}$ \hfill \ans{$0$ (row 2 $=2\times$ row 1)}
    \end{enumerate}
  \item \textbf{Volume from three columns.} Find the volume of the box spanned by the columns.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=5pt]
      \item $\begin{bmatrix} 4 \\ 0 \\ 0 \end{bmatrix}, \begin{bmatrix} 0 \\ 3 \\ 0 \end{bmatrix}, \begin{bmatrix} 0 \\ 0 \\ 2 \end{bmatrix}$ \hfill \ans{axis box: $4\cdot3\cdot2=24$}
      \item $\begin{bmatrix} 1 \\ 0 \\ 0 \end{bmatrix}, \begin{bmatrix} 2 \\ 3 \\ 0 \end{bmatrix}, \begin{bmatrix} 1 \\ 4 \\ 2 \end{bmatrix}$ \hfill \ans{triangular: $1\cdot3\cdot2=6$}
    \end{enumerate}
  \item \textbf{Zero determinant \& invertibility (§2).} Find $x$ so that
        $A=\begin{bmatrix} 2 & 1 & 0 \\ 1 & x & 1 \\ 0 & 1 & 2 \end{bmatrix}$ has $\det=0$.
        Is $A$ invertible for that $x$? What does $\det=0$ say about the three columns?
        \ansline{$\det A = 2(2x-1) - 1(2-0) + 0 = 4x-4$. Set $4x-4=0\Rightarrow x=1$. Not invertible at $x=1$: the three columns lie in one plane (the box is flat, zero volume).}
  \item \textbf{Explain.} For a $3\times3$ matrix, what does $|\det A|$ measure geometrically? What does
        $\det A=0$ tell you about the three columns and about the inverse?
        \ansline{$|\det A|$ is the \emph{volume} of the box the three columns span. $\det A=0$ means the columns lie in one plane (flat box, zero volume), so $A$ has \emph{no inverse}.}
\end{enumerate}
\end{notesbox}

\begin{extensionbox}
\textbf{Volume-scaling factor (a first look at transformations).} Applying a matrix $A$ to the
\textbf{unit cube} (volume $1$) produces the box spanned by $A$'s columns, whose volume is $|\det A|$.
So $|\det A|$ is the factor every volume is multiplied by (Lesson~5.3).
\begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=5pt]
  \item For $A=\begin{bmatrix} 3 & 0 & 0 \\ 0 & 2 & 0 \\ 0 & 0 & 2 \end{bmatrix}$, find the volume of the image of the unit cube. By what factor did volume grow?
  \item For $B=\begin{bmatrix} 2 & 0 & 0 \\ 0 & 3 & 0 \\ 0 & 0 & -1 \end{bmatrix}$, find $\det B$. What is the volume factor, and what does the sign say happened to the cube?
\end{enumerate}
\ansline{(a) $\det A=3\cdot2\cdot2=12$; image volume $=12$, so volume grew by a factor of $12$.}
\ansline{(b) $\det B=2\cdot3\cdot(-1)=-6$; volume factor $|-6|=6$, and the negative sign means the cube was \emph{flipped} (reflected).}
\end{extensionbox}

\begin{teachernote}
Item~1 mixes a full expansion (a), the triangular shortcut (b), and a $\det=0$ from dependent rows (c).
Item~2(b) is triangular, so volume is just the diagonal product --- reward students who notice instead
of expanding. Item~3 is the §2 callback (no inverse when $\det=0$), item~4 the target justification
(volume $+$ flat box $=$ no inverse). The extension is the Lesson~5.3 preview: $|\det|$ is the
volume-scaling factor and a negative determinant is a reflection. Most common slips: the middle sign
($-a_2$), arithmetic inside a minor, and dropping the absolute value for ``volume.''
\end{teachernote}
