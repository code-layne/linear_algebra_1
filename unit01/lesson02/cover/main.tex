  \item \ldots{} find the \textbf{angle} between two vectors with $\cos\theta=\dfrac{\mathbf{v}\cdot\mathbf{w}}{\|\mathbf{v}\|\,\|\mathbf{w}\|}$, and read the \textbf{sign} of the dot product.
  \item \ldots{} decide when two vectors are \textbf{perpendicular} ($\mathbf{v}\cdot\mathbf{w}=0$) and explain what that means about the angle.
\end{enumerate}

\begin{remindbox}
\small The \textbf{dot product} $\mathbf{v}\cdot\mathbf{w}=v_1w_1+v_2w_2$ turns two vectors into
one number. It recovers \textbf{length} through $\|\mathbf{v}\|=\sqrt{\mathbf{v}\cdot\mathbf{v}}$
and \textbf{angle} through $\cos\theta=\dfrac{\mathbf{v}\cdot\mathbf{w}}{\|\mathbf{v}\|\,\|\mathbf{w}\|}$.
A dot product of \textbf{zero} means the two vectors are \textbf{perpendicular}.
\end{remindbox}
