\begin{enumerate}[leftmargin=1.5em, itemsep=5pt]
  \item \ldots{} write one elimination step as an \textbf{elimination matrix} $E$, and check that $EA$ does the step.
  \item \ldots{} \textbf{undo} a step with $E^{-1}$ (add the multiplier back), knowing $E^{-1}E=I$.
  \item \ldots{} build a $2\times2$ \textbf{inverse} $A^{-1}$ --- by the $ad-bc$ formula and by \textbf{Gauss--Jordan} on $[\,A\mid I\,]$ --- and solve with $\mathbf{x}=A^{-1}\mathbf{b}$.
  \item \ldots{} recognize a \textbf{singular} matrix ($ad-bc=0$) as one with no inverse.
\end{enumerate}
\end{learningtargetbox}

\begin{remindbox}
\small An \textbf{elimination matrix} $E$ is the identity with $-\ell$ in one below-diagonal spot;
$EA$ does one elimination step. Its inverse $E^{-1}$ puts $+\ell$ there and \textbf{adds back} what
$E$ subtracted, so $E^{-1}E=I$. The \textbf{inverse} $A^{-1}$ satisfies $A^{-1}A=I$, so
$A\mathbf{x}=\mathbf{b}$ has the single solution $\mathbf{x}=A^{-1}\mathbf{b}$. For $2\times2$,
$A^{-1}=\frac{1}{ad-bc}\begin{bsmallmatrix} d&-b\\-c&a \end{bsmallmatrix}$; if $ad-bc=0$ the matrix
is \textbf{singular} --- no inverse.
\end{remindbox}
