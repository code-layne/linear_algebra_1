\[
  A\begin{bmatrix} 2 \\ 1 \end{bmatrix}
  = 2\begin{bmatrix} 1 \\ 2 \end{bmatrix} + 1\begin{bmatrix} 3 \\ 1 \end{bmatrix}
  = \blank{2.6cm}
