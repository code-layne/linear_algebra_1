\begin{headlinebox}{blush}
\small\textbf{Are These Viewers Alike?} A streaming app stores each viewer's taste as a vector
$\begin{bsmallmatrix} \text{action rating} \\ \text{romance rating} \end{bsmallmatrix}$, where
each rating runs from $-2$ (strongly dislike) to $+2$ (strongly like). Two viewers have
\emph{similar} taste when their vectors point the \emph{same way} --- and the \textbf{dot
product} measures exactly that. Work with your partner and \emph{show your steps}.
\end{headlinebox}

\vspace{0.10in}

\begin{tcolorbox}[colback=white, colframe=black!40, title=\textbf{Tier R --- Remediate},
    fonttitle=\bfseries, arc=1.5mm, left=3mm, right=3mm, top=2mm, bottom=2mm, breakable]
Three viewers: $\text{Ava}=\begin{bmatrix} 2 \\ 1 \end{bmatrix}$,
$\text{Ben}=\begin{bmatrix} 1 \\ 2 \end{bmatrix}$,
$\text{Cora}=\begin{bmatrix} -1 \\ 2 \end{bmatrix}$.
\begin{enumerate}[leftmargin=1.7em, itemsep=8pt]
  \item Compute $\text{Ava}\cdot\text{Ben}$.
\begin{work}
  \text{Ava}\cdot\text{Ben} &= (2)(1)+(1)(2) \\
                            &= 4
\end{work}
  \item Compute $\text{Ava}\cdot\text{Cora}$.
\begin{work}
  \text{Ava}\cdot\text{Cora} &= (2)(-1)+(1)(2) \\
                             &= 0
\end{work}
  \item Find the lengths $\|\text{Ava}\|$ and $\|\text{Ben}\|$ using $\|\vv\|=\sqrt{\vv\cdot\vv}$.
\begin{work}
  \|\text{Ava}\| &= \sqrt{2^2+1^2} \\
                 &= \sqrt5 \\
  \|\text{Ben}\| &= \sqrt{1^2+2^2} \\
                 &= \sqrt5
\end{work}
  \item Which of the two dot products in (1) and (2) is bigger?
        \ansline{(1) is bigger: $4>0$. Ava and Ben point more the same way than Ava and Cora.}
\end{enumerate}
\end{tcolorbox}

\boxguard[20]
\begin{tcolorbox}[colback=white, colframe=black!40, title=\textbf{Tier A --- Approaching Proficiency},
    fonttitle=\bfseries, arc=1.5mm, left=3mm, right=3mm, top=2mm, bottom=2mm, breakable]
Same viewers, plus $\text{Dan}=\begin{bmatrix} -2 \\ -1 \end{bmatrix}$. Use
$\cos\theta=\dfrac{\vv\cdot\ww}{\|\vv\|\,\|\ww\|}$; round angles to the nearest degree.
\begin{enumerate}[leftmargin=1.7em, itemsep=8pt]
  \item Find the angle between Ava and Ben. Are their tastes similar?
\begin{work}
  \cos\theta &= \frac{\text{Ava}\cdot\text{Ben}}{\|\text{Ava}\|\,\|\text{Ben}\|} \\
             &= \frac{4}{\sqrt5\cdot\sqrt5} \\
             &= 0.8 \\
      \theta &\approx 37^\circ
\end{work}
        \ansline{A small angle --- yes, their tastes are similar.}
  \item Compute $\text{Ava}\cdot\text{Cora}$ and $\text{Ava}\cdot\text{Dan}$. From the \emph{sign}
        of each, describe how each viewer's taste compares to Ava's (similar / unrelated / opposite).
        \ansline{$\text{Ava}\cdot\text{Cora}=0$ (right angle) $\Rightarrow$ \emph{unrelated} taste.
        $\text{Ava}\cdot\text{Dan}=(2)(-2)+(1)(-1)=-5<0$ (obtuse angle) $\Rightarrow$ \emph{opposite} taste.}
  \item \textbf{Interpret.} Cora's dot product with Ava is $0$. What does a right angle mean for
        two viewers' tastes?
        \ansline{Their tastes are independent --- neither aligned nor opposed. Knowing Ava's
        preferences tells you nothing about Cora's.}
\end{enumerate}
\end{tcolorbox}

\begin{tcolorbox}[colback=white, colframe=black!40, title=\textbf{Tier E --- Extension},
    fonttitle=\bfseries, arc=1.5mm, left=3mm, right=3mm, top=2mm, bottom=2mm, breakable]
\begin{enumerate}[leftmargin=1.7em, itemsep=8pt]
  \item Find a taste vector that is \textbf{perpendicular} to $\text{Ava}=\begin{bsmallmatrix} 2\\1 \end{bsmallmatrix}$
        (a nonzero vector whose dot product with Ava is $0$). Show it works.
        \ansline{Need $2x+y=0$. For example $\begin{bsmallmatrix} 1\\-2 \end{bsmallmatrix}$:
        $(2)(1)+(1)(-2)=0$. \ (Any nonzero multiple of $\begin{bsmallmatrix} 1\\-2 \end{bsmallmatrix}$ works.)}
  \item A new viewer's taste is $3\,\text{Ava}=\begin{bsmallmatrix} 6\\3 \end{bsmallmatrix}$. Find the
        angle between Ava and $3\,\text{Ava}$. What does this say about length versus direction?
        \ansline{$\cos\theta=\dfrac{(2)(6)+(1)(3)}{\sqrt5\cdot\sqrt{45}}=\dfrac{15}{15}=1$, so $\theta=0^\circ$.
        Scaling changes a vector's length but \emph{not} its direction --- same taste, ``bigger'' ratings.}
  \item \textbf{Justify.} Explain, using the angle formula, why $\vv\cdot\ww=0$ always forces a
        right angle between two nonzero vectors.
        \ansline{$\cos\theta=\dfrac{\vv\cdot\ww}{\|\vv\|\,\|\ww\|}$. If $\vv\cdot\ww=0$ and both
        vectors are nonzero, the numerator is $0$ and the denominator is not, so $\cos\theta=0$.
        The only angle between $0^\circ$ and $180^\circ$ with cosine $0$ is $90^\circ$.}
\end{enumerate}
\end{tcolorbox}
