% COFACTOR EXPANSION
\begin{frame}[t, plain]
  \burgundyheader{Cofactor expansion: three 2$\times$2s you know}
  \sectionlabel[goldacc]{Method}
  \begin{itemize}
    \setlength{\itemsep}{5pt}
    \item Go across the top row. Cross out each entry's row \& column (its \textbf{minor}); attach signs $+\,-\,+$:
      \[ \det\begin{bsmallmatrix} a_1 & a_2 & a_3\\ b_1 & b_2 & b_3\\ c_1 & c_2 & c_3 \end{bsmallmatrix}
        = a_1\det\begin{bsmallmatrix} b_2 & b_3\\ c_2 & c_3 \end{bsmallmatrix}
        - a_2\det\begin{bsmallmatrix} b_1 & b_3\\ c_1 & c_3 \end{bsmallmatrix}
        + a_3\det\begin{bsmallmatrix} b_1 & b_2\\ c_1 & c_2 \end{bsmallmatrix}. \]
    \item \textbf{Worked:} $\det\begin{bsmallmatrix} 2 & 1 & 0\\ 1 & 3 & 1\\ 0 & 1 & 2 \end{bsmallmatrix}
          = 2(5) - 1(2) + 0 = 8.$ \; (Top-row $0$ kills a term --- expand where the zeros are.)
  \end{itemize}
\end{frame}

% VOLUME + TRIANGULAR
\begin{frame}[t, plain]
  \burgundyheader{Volume, and a triangular shortcut}
  \sectionlabel{Geometry}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      $|\det A|$ is the \textbf{volume} of the box.\\[6pt]
      \textbf{Axis box:} $\det\,\mathrm{diag}(2,3,4)=24=2\!\cdot\!3\!\cdot\!4.$\\[6pt]
      \textbf{Triangular} (all zeros below the diagonal): $\det$ is the \emph{product of the diagonal}.
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering
      \[ \det\begin{bsmallmatrix} 2 & 5 & 1\\ 0 & 3 & 4\\ 0 & 0 & 2 \end{bsmallmatrix} = 2\cdot3\cdot2 = 12. \]
      \vspace{4pt}
      No expansion needed --- just multiply the diagonal.
    \end{column}
  \end{columns}
\end{frame}

% ZERO + ROAD AHEAD
\begin{frame}[t, plain]
  \burgundyheader{Zero volume means flat --- and no inverse}
  \sectionlabel[goldacc]{Collapse}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item \textbf{Coplanar columns $\Rightarrow$ flat box $\Rightarrow$ zero volume.}
          $\det\begin{bsmallmatrix} 1 & 0 & 1\\ 0 & 1 & 1\\ 1 & 1 & 2 \end{bsmallmatrix}=0$
          (column~3 $=$ column~1 $+$ column~2).
    \item \textbf{The whole story:} \; $\det A\ne0 \;\Leftrightarrow\;$ columns independent
          $\;\Leftrightarrow\; A$ is \textbf{invertible}.
    \item \textbf{Sign $=$ orientation:} swapping two columns negates $\det$ (a mirror flip).
  \end{itemize}
  \vspace{2pt}
  \begin{block}{The road ahead}
    \textbf{5.2} properties \& applications of determinants \;$\bullet$\; \textbf{5.3} transformations,
    where $|\det A|$ is the \emph{volume-scaling factor}.
  \end{block}
\end{frame}
