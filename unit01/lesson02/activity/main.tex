\begin{headlinebox}{blush}
\small\textbf{Are These Viewers Alike?} A streaming app stores each viewer's taste as a vector
$\begin{bsmallmatrix} \text{action rating} \\ \text{romance rating} \end{bsmallmatrix}$, where
each rating runs from $-2$ (strongly dislike) to $+2$ (strongly like). Two viewers have
\emph{similar} taste when their vectors point the \emph{same way} --- and the \textbf{dot
product} measures exactly that. Work with your partner and \emph{show your steps}.
\end{headlinebox}

\vspace{0.10in}

\begin{tcolorbox}[colback=white, colframe=black!40, title=\textbf{Tier R --- Remediate},
    fonttitle=\bfseries, arc=1.5mm, left=3mm, right=3mm, top=2mm, bottom=2mm, breakable]
Three viewers: $\text{Ava}=\begin{bmatrix} 2 \\ 1 \end{bmatrix}$,
$\text{Ben}=\begin{bmatrix} 1 \\ 2 \end{bmatrix}$,
$\text{Cora}=\begin{bmatrix} -1 \\ 2 \end{bmatrix}$.
\begin{enumerate}[leftmargin=1.7em, itemsep=10pt]
  \item Compute $\text{Ava}\cdot\text{Ben}$. \vspace{0.4cm}
  \item Compute $\text{Ava}\cdot\text{Cora}$. \vspace{0.4cm}
  \item Find the lengths $\|\text{Ava}\|$ and $\|\text{Ben}\|$ using $\|\vv\|=\sqrt{\vv\cdot\vv}$. \vspace{0.5cm}
  \item Which of the two dot products in (1) and (2) is bigger? \writeline
\end{enumerate}
\end{tcolorbox}

\begin{tcolorbox}[colback=white, colframe=black!40, title=\textbf{Tier A --- Approaching Proficiency},
    fonttitle=\bfseries, arc=1.5mm, left=3mm, right=3mm, top=2mm, bottom=2mm, breakable]
Same viewers, plus $\text{Dan}=\begin{bmatrix} -2 \\ -1 \end{bmatrix}$. Use
$\cos\theta=\dfrac{\vv\cdot\ww}{\|\vv\|\,\|\ww\|}$; round angles to the nearest degree.
\begin{enumerate}[leftmargin=1.7em, itemsep=10pt]
  \item Find the angle between Ava and Ben. Are their tastes similar? \vspace{0.9cm}
  \item Compute $\text{Ava}\cdot\text{Cora}$ and $\text{Ava}\cdot\text{Dan}$. From the \emph{sign}
        of each, describe how each viewer's taste compares to Ava's (similar / unrelated / opposite). \writelines{2}
  \item \textbf{Interpret.} Cora's dot product with Ava is $0$. What does a right angle mean for
        two viewers' tastes? \writelines{2}
\end{enumerate}
\end{tcolorbox}

\begin{tcolorbox}[colback=white, colframe=black!40, title=\textbf{Tier E --- Extension},
    fonttitle=\bfseries, arc=1.5mm, left=3mm, right=3mm, top=2mm, bottom=2mm, breakable]
\begin{enumerate}[leftmargin=1.7em, itemsep=10pt]
  \item Find a taste vector that is \textbf{perpendicular} to $\text{Ava}=\begin{bsmallmatrix} 2\\1 \end{bsmallmatrix}$
        (a nonzero vector whose dot product with Ava is $0$). Show it works. \writelines{2}
  \item A new viewer's taste is $3\,\text{Ava}=\begin{bsmallmatrix} 6\\3 \end{bsmallmatrix}$. Find the
        angle between Ava and $3\,\text{Ava}$. What does this say about length versus direction? \writelines{2}
  \item \textbf{Justify.} Explain, using the angle formula, why $\vv\cdot\ww=0$ always forces a
        right angle between two nonzero vectors. \writelines{3}
\end{enumerate}
\end{tcolorbox}
