  \item \textbf{Solve by elimination:} $\quad 2x + y = 5,\qquad 4x + 3y = 13.$
        \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=7pt]
          \item Name the \textbf{pivot} and the \textbf{multiplier} $\ell$ that clears the entry below it.
                \hfill pivot $=\blank{1.4cm}$\quad $\ell=\blank{1.4cm}$
          \item Do the elimination step, write the upper-triangular system, and back-substitute.
\begin{work}
  (4x+3y)-2(2x+y) &= 13-10 \\
                y &= 3 \\
             2x+3 &= 5 \\
                x &= 1
\end{work}
          \item \textbf{Check} by rebuilding $\bb$. \hfill $x=\blank{1.4cm}$\quad $y=\blank{1.4cm}$
\begin{work}
  1\begin{bsmallmatrix} 2\\4 \end{bsmallmatrix} + 3\begin{bsmallmatrix} 1\\3 \end{bsmallmatrix}
      &= \begin{bsmallmatrix} 5\\13 \end{bsmallmatrix} \ \ \text{\checkmark}
\end{work}
        \end{enumerate}
  \item \textbf{Zero pivot.} Priya starts elimination on
        $\ 0x + 3y = 9,\ \ x + 2y = 8\ $ and gets stuck: there is no pivot to work with in the top
        row. What move fixes this, \emph{why} does it work, and what is the solution?
\begin{work}
  \text{swap rows:}\quad x+2y &= 8 \\
                          3y &= 9 \\
                           y &= 3 \\
                           x &= 2
\end{work}
        \writelines{2}
\end{enumerate}
